#include <iostream>
#include <fstream>
#include <vector>
#include <thread>
#include <zlib.h>

const int CHUNK_SIZE = 262144; // 256 KB

void compress_chunk(const std::vector<char>& input, std::vector<char>& output, size_t chunk_index) {
    uLongf compressed_size = compressBound(input.size());
    output.resize(compressed_size);
    
    if (compress(reinterpret_cast<Bytef*>(&output[0]), &compressed_size, 
                 reinterpret_cast<const Bytef*>(&input[0]), input.size()) != Z_OK) {
        std::cerr << "Compression failed for chunk " << chunk_index << std::endl;
        return;
    }
    output.resize(compressed_size);
}

void decompress_chunk(const std::vector<char>& input, std::vector<char>& output, size_t original_size) {
    output.resize(original_size);
    
    if (uncompress(reinterpret_cast<Bytef*>(&output[0]), &original_size, 
                   reinterpret_cast<const Bytef*>(&input[0]), input.size()) != Z_OK) {
        std::cerr << "Decompression failed." << std::endl;
    }
}

void compress_file(const std::string& input_file, const std::string& output_file) {
    std::ifstream in(input_file, std::ios::binary);
    std::ofstream out(output_file, std::ios::binary);
    
    if (!in || !out) {
        std::cerr << "Failed to open files." << std::endl;
        return;
    }
    
    std::vector<std::thread> threads;
    std::vector<std::vector<char>> compressed_chunks;
    std::vector<std::vector<char>> input_chunks;
    
    char buffer[CHUNK_SIZE];
    while (in.read(buffer, CHUNK_SIZE) || in.gcount() > 0) {
        std::vector<char> input_data(buffer, buffer + in.gcount());
        std::vector<char> compressed_data;
        input_chunks.push_back(input_data);
        compressed_chunks.emplace_back();
        threads.emplace_back(compress_chunk, std::cref(input_data), std::ref(compressed_chunks.back()), input_chunks.size() - 1);
    }
    
    for (auto& t : threads) t.join();
    
    for (const auto& chunk : compressed_chunks) {
        size_t size = chunk.size();
        out.write(reinterpret_cast<const char*>(&size), sizeof(size));
        out.write(chunk.data(), chunk.size());
    }
}

void decompress_file(const std::string& input_file, const std::string& output_file) {
    std::ifstream in(input_file, std::ios::binary);
    std::ofstream out(output_file, std::ios::binary);
    
    if (!in || !out) {
        std::cerr << "Failed to open files." << std::endl;
        return;
    }
    
    std::vector<std::thread> threads;
    std::vector<std::vector<char>> decompressed_chunks;
    std::vector<std::vector<char>> compressed_chunks;
    
    while (!in.eof()) {
        size_t chunk_size;
        in.read(reinterpret_cast<char*>(&chunk_size), sizeof(chunk_size));
        if (in.eof()) break;
        
        std::vector<char> compressed_data(chunk_size);
        in.read(compressed_data.data(), chunk_size);
        compressed_chunks.push_back(compressed_data);
        decompressed_chunks.emplace_back();
        threads.emplace_back(decompress_chunk, std::cref(compressed_data), std::ref(decompressed_chunks.back()), CHUNK_SIZE);
    }
    
    for (auto& t : threads) t.join();
    
    for (const auto& chunk : decompressed_chunks) {
        out.write(chunk.data(), chunk.size());
    }
}

int main(int argc, char* argv[]) {
    if (argc != 4) {
        std::cerr << "Usage: " << argv[0] << " <compress|decompress> <input_file> <output_file>" << std::endl;
        return 1;
    }
    
    std::string mode = argv[1];
    if (mode == "compress") {
        compress_file(argv[2], argv[3]);
        std::cout << "Compression completed." << std::endl;
    } else if (mode == "decompress") {
        decompress_file(argv[2], argv[3]);
        std::cout << "Decompression completed." << std::endl;
    } else {
        std::cerr << "Invalid mode. Use 'compress' or 'decompress'." << std::endl;
        return 1;
    }
    return 0;
}
